Question 14: Write a function that takes an array of numbers and returns the sum of all elements.

code:

function sumArray(numbers) {
    let sum = 0;

    for (let i = 0; i < numbers.length; i++) {
        sum = sum + numbers[i];
    }

    return sum;
}

let numbers = [10, 20, 30, 40, 50];

console.log(sumArray(numbers));

output

150
